  // attitude.h
  #pragma once
  #include "types.h"

  namespace smas {

  struct AttitudeState {
      double q0 = 1.0, q1 = 0.0, q2 = 0.0, q3 = 0.0;  // quaternion
      double wx = 0.0, wy = 0.0, wz = 0.0;              // angular vel (rad/s)
      double wheel_h[3] = {0, 0, 0};                     // wheel momentum (Nms)
      double sun_angle = 0.0;     // angle between panel normal and sun (rad)
      double nadir_error = 0.0;   // pointing error from nadir (rad)
  };

  class AttitudeModel {
  public:
      void reset();
      void step(double dt, const Vec3& pos_eci, const Vec3& sun_dir,
                const double* wheel_torque_cmd);  // 3-axis torque command

      AttitudeState state;

      // Derived
      double cos_sun_angle() const;   // for solar power
      bool is_nadir_pointing(double tol_deg = 5.0) const;

  private:
      void compute_gravity_gradient_torque(const Vec3& pos_eci);
      void compute_srp_torque(const Vec3& sun_dir);
      void integrate_quaternion(double dt);
      void update_wheels(double dt, const double* cmd);

      Vec3 gg_torque_{0,0,0};
      Vec3 srp_torque_{0,0,0};
  };

  }  // namespace smas

  // attitude.cpp — Key equations:
  //
  // Quaternion kinematics:
  //   q_dot = 0.5 * q ⊗ [0, wx, wy, wz]
  //   q_new = normalize(q + q_dot * dt)
  //
  // Gravity gradient torque (simplified):
  //   T_gg = (3 * mu / r^5) * (r × (I · r))
  //   Approximate: T_gg ≈ 3*mu/r^3 * (Iz-Iy) * sin(2θ) / 2
  //   For PROBA-1: ~10^-7 Nm (very small)
  //
  // Reaction wheel:
  //   h_dot = torque_cmd - T_external
  //   Saturation: |h| < H_MAX (typically 0.1-1.0 Nms for small sat)
  //
  // Sun angle:
  //   panel_normal = attitude_matrix * [0, 0, 1]  (body Z = panel normal)
  //   sun_angle = acos(dot(panel_normal, sun_dir))
  //
  // Nadir error:
  //   nadir_dir = -normalize(pos_eci)
  //   body_z = attitude_matrix * [0, 0, 1]
  //   nadir_error = acos(dot(body_z, nadir_dir))

  // Main loop: step all 4
  bool constellationDone = false;
  while (!constellationDone) {
      for (int i = 0; i < NUM_SATS; i++) {
          if (engines[i].IsDone) continue;
          var obs = obsBuilders[i].Build(engines[i].State);
          var action = inference.Run(obs);
          action = governors[i].Apply(action, engines[i].State);
          engines[i].Step(action);
      }
      constellationDone = engines.All(e => e.IsDone);
      // Send telemetry for all 4
  }
